<script>

let projects =
JSON.parse(localStorage.getItem("projects")) || [];

let items =
JSON.parse(localStorage.getItem("items")) || [];

let quoteTotal = 0;

function showPage(page){

document
.querySelectorAll("#dashboard,#project,#price,#quote")
.forEach(x=>x.classList.add("hidden"));

document
.getElementById(page)
.classList.remove("hidden");

}

function saveProject(){

let data={
projectId:
document.getElementById("projectId").value,

customer:
document.getElementById("customerName").value,

contact:
document.getElementById("contact").value,

address:
document.getElementById("address").value,

title:
document.getElementById("projectTitle").value
};

projects.push(data);

localStorage.setItem(
"projects",
JSON.stringify(projects)
);

document.getElementById(
"totalProjects"
).innerText=projects.length;

alert("Project Saved");

}

function saveItem(){

let item={

code:
document.getElementById("itemCode").value,

description:
document.getElementById("itemDescription").value,

unit:
document.getElementById("itemUnit").value,

rate:
document.getElementById("itemRate").value

};

items.push(item);

localStorage.setItem(
"items",
JSON.stringify(items)
);

loadPriceTable();

loadDropdown();

alert("Item Saved");

}

function loadPriceTable(){

let table =
document.getElementById("priceTable");

table.innerHTML=`

<tr>
<th>Code</th>
<th>Description</th>
<th>Rate</th>
</tr>

`;

items.forEach(i=>{

table.innerHTML+=`

<tr>

<td>${i.code}</td>

<td>${i.description}</td>

<td>${i.rate}</td>

</tr>

`;

});

document.getElementById(
"totalItems"
).innerText=items.length;

}

function loadDropdown(){

let select=
document.getElementById("quoteItem");

select.innerHTML=
'<option value="">Select Item</option>';

items.forEach((i,index)=>{

select.innerHTML+=
`<option value="${index}">
${i.code}
</option>`;

});

}

function loadItem(){

let index=
document.getElementById("quoteItem").value;

if(index==="") return;

let item=items[index];

document.getElementById(
"quoteDescription"
).value=item.description;

document.getElementById(
"quoteUnit"
).value=item.unit;

document.getElementById(
"quoteRate"
).value=item.rate;

}

function calculateAmount(){

let qty=
Number(document.getElementById(
"quoteQty"
).value);

let rate=
Number(document.getElementById(
"quoteRate"
).value);

document.getElementById(
"quoteAmount"
).value=(qty*rate).toFixed(2);

}

function addQuoteItem(){

let desc=
document.getElementById(
"quoteDescription"
).value;

let qty=
document.getElementById(
"quoteQty"
).value;

let rate=
document.getElementById(
"quoteRate"
).value;

let amount=
document.getElementById(
"quoteAmount"
).value;

let table=
document.getElementById(
"quoteTable"
);

table.innerHTML+=`

<tr>

<td>${desc}</td>

<td>${qty}</td>

<td>${rate}</td>

<td>${amount}</td>

</tr>

`;

quoteTotal += Number(amount);

document.getElementById(
"grandTotal"
).innerText=
quoteTotal.toFixed(2);

}

loadPriceTable();
loadDropdown();

document.getElementById(
"totalProjects"
).innerText=
projects.length;
